cmake_minimum_required(VERSION 3.20)
project(DistributedMediaStorage LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "TNV requires Linux; configure in an Ubuntu local build directory")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build directory")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(ACL_ROOT "" CACHE PATH "ACL installation prefix")
find_path(ACL_CPP_INCLUDE_DIR lib_acl.hpp HINTS ${ACL_ROOT} PATH_SUFFIXES include/acl-lib/acl_cpp include/acl_cpp include)
find_path(ACL_INCLUDE_DIR lib_acl.h HINTS ${ACL_ROOT} PATH_SUFFIXES include/acl-lib/acl include/acl include)
find_library(ACL_LIBRARY NAMES acl_all HINTS ${ACL_ROOT} PATH_SUFFIXES lib lib64)
find_path(MYSQL_INCLUDE_DIR mysql.h PATH_SUFFIXES mysql mariadb)
find_library(MYSQL_LIBRARY NAMES mysqlclient mariadb)
foreach(dep ACL_CPP_INCLUDE_DIR ACL_INCLUDE_DIR ACL_LIBRARY MYSQL_INCLUDE_DIR MYSQL_LIBRARY)
  if(NOT ${dep})
    message(FATAL_ERROR "Missing ${dep}. Install dependencies manually; set ACL_ROOT or the explicit include/library cache paths.")
  endif()
endforeach()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(tnv_common STATIC src/01_common/04_util.cpp)
target_include_directories(tnv_common PUBLIC src/01_common ${ACL_CPP_INCLUDE_DIR} ${ACL_INCLUDE_DIR} ${MYSQL_INCLUDE_DIR})
target_link_libraries(tnv_common PUBLIC ${ACL_LIBRARY} ${MYSQL_LIBRARY} OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(tnv_common PRIVATE -Wall -Wextra -Wpedantic)
foreach(pair "id;03_id" "storage;04_storage")
  list(GET pair 0 name)
  list(GET pair 1 dir)
  file(GLOB sources CONFIGURE_DEPENDS "src/${dir}/*.cpp")
  add_executable(${name} ${sources})
  target_link_libraries(${name} PRIVATE tnv_common)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
file(GLOB client_sources CONFIGURE_DEPENDS src/05_client/*.cpp)
list(REMOVE_ITEM client_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/05_client/09_main.cpp")
add_library(client_static STATIC ${client_sources} src/01_common/04_util.cpp)
set_target_properties(client_static PROPERTIES OUTPUT_NAME client)
target_link_libraries(client_static PUBLIC tnv_common)
target_compile_options(client_static PRIVATE -Wall -Wextra -Wpedantic)
add_executable(client src/05_client/09_main.cpp)
target_link_libraries(client PRIVATE client_static)
target_compile_options(client PRIVATE -Wall -Wextra -Wpedantic)

set(HIGHPERFRPC_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/HighPerfRPC" CACHE PATH "HighPerfRPC source directory")
option(BUILD_TESTING "Build tests" OFF)
option(HIGHPERFRPC_BUILD_RPC "Build framework RPC" ON)
option(HIGHPERFRPC_BUILD_EXAMPLES "Build framework examples" OFF)
option(HIGHPERFRPC_BUILD_BENCHMARKS "Build framework benchmarks" OFF)
option(HIGHPERFRPC_ENABLE_ASAN "Enable sanitizer" OFF)
find_package(Protobuf REQUIRED MODULE)
add_subdirectory("${HIGHPERFRPC_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/highperfrpc")
set(tnv_generated "${CMAKE_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${tnv_generated}/tnv/control.pb.cc" "${tnv_generated}/tnv/control.pb.h"
 COMMAND ${CMAKE_COMMAND} -E make_directory "${tnv_generated}/tnv"
 COMMAND protobuf::protoc --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto --cpp_out=${tnv_generated} ${CMAKE_CURRENT_SOURCE_DIR}/proto/tnv/control.proto
 DEPENDS proto/tnv/control.proto protobuf::protoc VERBATIM)
add_custom_target(tnv_proto_generate DEPENDS "${tnv_generated}/tnv/control.pb.cc" "${tnv_generated}/tnv/control.pb.h")
add_library(tnv_proto STATIC "${tnv_generated}/tnv/control.pb.cc")
add_dependencies(tnv_proto tnv_proto_generate)
target_include_directories(tnv_proto PUBLIC "${tnv_generated}")
target_link_libraries(tnv_proto PUBLIC protobuf::libprotobuf)
add_library(tnv_rpc_client STATIC src/rpc_control/client.cpp)
add_dependencies(tnv_rpc_client tnv_proto_generate)
target_include_directories(tnv_rpc_client PUBLIC src/rpc_control)
target_link_libraries(tnv_rpc_client PUBLIC tnv_proto highperfrpc_rpc)
target_link_libraries(client_static PUBLIC tnv_rpc_client)
target_link_libraries(storage PRIVATE tnv_rpc_client)
add_dependencies(client_static tnv_proto_generate)
add_dependencies(client tnv_proto_generate)
add_dependencies(storage tnv_proto_generate)
add_executable(tracker src/rpc_control/tracker.cpp src/02_tracker/02_globals.cpp src/02_tracker/04_cache.cpp src/02_tracker/06_db.cpp)
add_dependencies(tracker tnv_proto_generate)
target_link_libraries(tracker PRIVATE tnv_common tnv_rpc_client)
add_executable(tnv_bench src/bench/main.cpp)
add_dependencies(tnv_bench tnv_proto_generate)
target_link_libraries(tnv_bench PRIVATE tnv_rpc_client)
foreach(t tnv_rpc_client tracker tnv_bench)
 target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
add_executable(tnv_file_bench src/bench/file_main.cpp)
target_include_directories(tnv_file_bench PRIVATE src/05_client)
target_link_libraries(tnv_file_bench PRIVATE client_static)
add_dependencies(tnv_file_bench tnv_proto_generate)
target_compile_options(tnv_file_bench PRIVATE -Wall -Wextra -Wpedantic)
